  1173→function buildStages(upTo: PipelineStageName): PipelineStage[] {
  1174→  const order: PipelineStageName[] = [
  1175→    "queued",
  1176→    "uploading",
  1177→    "parsing",
  1178→    "themes",
  1179→    "swot",
  1180→    "strategy",
  1181→    "campaigns",
  1182→    "mongo",
  1183→    "done",
  1184→  ];
  1185→  const idx = order.indexOf(upTo);
  1186→  return order.map((name, i) => {
  1187→    const status: PipelineStage["status"] =
  1188→      i < idx ? "completed" : i === idx ? "running" : "pending";
  1189→    return {
  1190→      name,
  1191→      label: STAGE_META[name],
  1192→      status,
  1193→      started_at: i <= idx ? iso(1000 * 60 * (idx - i)) : undefined,
  1194→      completed_at: i < idx ? iso(1000 * 60 * (idx - i - 1)) : undefined,
  1195→    };
  1196→  });
  1197→}
